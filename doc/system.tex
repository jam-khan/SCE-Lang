\section{$\lambda_E^{\leq}$ (Seal)}

\subsection{Syntax}

\[
\begin{array}{r@{\;}c@{\;}l@{\qquad}l}
A,B,C,R,S,T,\Gamma &::=& \Int \mid \tp \mid A \to B \mid A \,\&\, B \mid \{\ell : A\} \mid \bra{n} & \\
  &\mid& A \vee B \mid X_k \mid \mu T \mid \bv{i} \mid \allb{A} & \\
\Delta &\in& \mathbb{N} \rightharpoonup \mathsf{Typ} \qquad \emptyset \;=\; \lambda n.\ \mathsf{none} & \\[2pt]
e &::=& \qry \mid e.k \mid i \mid \tp \mid \lambda(A{\to}B).\,e \mid e_1 \bx e_2 \mid \langle e_1, \lambda(A{\to}B).\,e_2\rangle \mid e_1\,e_2 & \\
  &\mid& e_1 \mrg e_2 \mid \{\ell = e\} \mid e.\ell \mid (e : A) \mid \wrp{n}{e} \mid \sealE{n}{R}{S}{e} \mid \unsealE{n}{R}{S}{e} & \\
  &\mid& \mathsf{inl}^{B}\,e \mid \mathsf{inr}^{A}\,e \mid \mathsf{case}\ e\ \mathsf{of}\ e_1 \mid e_2
    \mid \mathsf{fold}^{T} e \mid \mathsf{unfold}\ e & \\
  &\mid& \mu\lambda(A{\to}B).\,e \mid \fclosx{e_1}{A}{B}{e_2}^{B_x}
    \mid \blam{A}{e} \mid \langle e_1, \blam{A}{e_2}\rangle \mid \bapp{e}{n} & \\[2pt]
v &::=& i \mid \tp \mid \langle v, \lambda(A{\to}B).\,e\rangle \mid \{\ell = v\} \mid v_1 \mrg v_2 \mid \wrp{n}{v} & \\
  &\mid& \mathsf{inl}^{B}\,v \mid \mathsf{inr}^{A}\,v \mid \mathsf{fold}^{T} v & \\
  &\mid& \fclosx{v}{A}{B}{e}^{B_x} \mid \langle v, \blam{A}{e}\rangle &
\end{array}
\]
$X_k$ is a $\mu$-bound type variable and $\bv{i}$ a $\forall$-bound \emph{brand}
variable (both de Bruijn, disjoint namespaces from each other and from concrete brands
$\bra{n}$); $\blam{A}{e}$ and its closure carry the $\forall$-body annotation $A$
(casting at a $\forall$-type is an identity that the value must pin).

\smallskip
Three substitutions, all shift-free because the substitutend is closed in the relevant
namespace: brand substitution $\subs{S}{n}{R}$ (replace $\bra{n}$ by $R$),
$\mu$-unfolding $\subs{T}{X_d}{S}^{\mu}$ (replace $X_d$ by $S$; $d{+}1$ under $\mu$),
and $\forall\beta$-instantiation $\subs{A}{\bv{d}}{\bra{n}}$ (replace $\bv{d}$ by the
\emph{concrete} $\bra{n}$; $d{+}1$ under $\forall$; $\mu$ binds $X$, not $\beta$ — no
bump).  Each is homomorphic on all other formers; the interesting leaves:
\[
\begin{array}{l}
\subs{\bra{m}}{n}{R} = \begin{cases} R & m = n \\ \bra{m} & m \neq n\end{cases} \qquad
\subs{X_k}{n}{R} = X_k \qquad
\subs{\bv{i}}{n}{R} = \bv{i} \\[3pt]
\subs{(\mu T)}{n}{R} = \mu\,\subs{T}{n}{R} \qquad
\subs{(\allb{A})}{n}{R} = \allb{\subs{A}{n}{R}} \\[3pt]
\subs{X_k}{X_d}{S}^{\mu} = \begin{cases} S & k = d \\ X_k & k \neq d\end{cases} \qquad
\subs{\bv{i}}{X_d}{S}^{\mu} = \bv{i} \qquad
\subs{\bra{m}}{X_d}{S}^{\mu} = \bra{m} \\[3pt]
\subs{\bv{i}}{\bv{d}}{\bra{n}} = \begin{cases} \bra{n} & i = d \\ \bv{i} & i \neq d\end{cases} \qquad
\subs{X_k}{\bv{d}}{\bra{n}} = X_k \\[3pt]
\subs{(\mu T)}{\bv{d}}{\bra{n}} = \mu\,\subs{T}{\bv{d}}{\bra{n}} \qquad
\subs{(\allb{A})}{\bv{d}}{\bra{n}} = \allb{\subs{A}{\bv{d{+}1}}{\bra{n}}}
\end{array}
\]
$\subs{e}{\bv{d}}{\bra{n}}$ pushes the instantiation into every type annotation of $e$
($d{+}1$ under $\Lambda$ and its closure's annotation/body; a captured environment sits
outside the binder and substitutes at $d$).

\smallskip
Occurrence predicates: $\hbv{T}$ (some $\bv{i}$ occurs in $T$, index-blind);
$\hallb{T}$ (some $\forall$ occurs in $T$); $\bcl{c}{T}$ (every $\bv{i}$ in $T$ is
bound below the cutoff: $i < c$ plus the enclosing binders; $c{+}1$ under $\forall$).
Size measures: $\mathsf{tsize}$ on types and $\mathsf{esize}$ on terms (annotations
excluded); both are preserved by $\forall\beta$-instantiation, which only trades a
$\bv{i}$ leaf for a $\bra{n}$ leaf — the well-foundedness of the logical relation and
of the fundamental lemma's induction rests on this.

\subsubsection*{Subtyping \quad $A <: B$}
\begin{mathpar}
\inferrule*[right=sint]{ }{\Int <: \Int}
\and
\inferrule*[right=stop]{ }{A <: \tp}
\and
\inferrule*[right=sarr]{B_1 <: A_1 \\ A_2 <: B_2}{A_1 \to A_2 <: B_1 \to B_2}
\and
\inferrule*[right=sandl]{A_1 <: A_3}{A_1 \& A_2 <: A_3}
\and
\inferrule*[right=sandr]{A_2 <: A_3}{A_1 \& A_2 <: A_3}
\and
\inferrule*[right=sand]{A_1 <: A_2 \\ A_1 <: A_3}{A_1 <: A_2 \& A_3}
\and
\inferrule*[right=srcd]{A <: B}{\{\ell : A\} <: \{\ell : B\}}
\and
\inferrule*[right=sbrand]{ }{\bra{n} <: \bra{n}}
\and
\inferrule*[right=sor]{A_1 <: B_1 \\ A_2 <: B_2}{A_1 \vee A_2 <: B_1 \vee B_2}
\and
\inferrule*[right=svar]{ }{X_k <: X_k}
\and
\inferrule*[right=smu]{ }{\mu T <: \mu T}
\and
\inferrule*[right=sbvar]{ }{\bv{i} <: \bv{i}}
\and
\inferrule*[right=sallb]{ }{\allb{A} <: \allb{A}}
\end{mathpar}
Unions are component-wise only (no injection $A <: A \vee B$: casting would have to
invent a tag); $\mu$, $X$, $\beta$ and $\forall$ are opaque, reflexive-only.
\emph{No} rule $\bra{n} <: R$: that would be translucency and kills representation
independence.

\and
\inferrule*[right=cbrand]{ }{\bra{n} \sqcap \bra{n}}
\and

\inferrule*[right=cvar]{ }{X_k \sqcap X_k}
\and

\and
\inferrule*[right=cbvar]{ }{\bv{i} \sqcap \bv{j}}
\and
\inferrule*[right=cbvarbrand]{ }{\bv{i} \sqcap \bra{n}}
\and
\inferrule*[right=cbrandbvar]{ }{\bra{n} \sqcap \bv{i}}

\and
\inferrule*[right=callbbv]{\hbv{A} \vee \hbv{A'}}{\allb{A} \sqcap \allb{A'}}
\and
\inferrule*[right=cmubv]{\hbv{T} \vee \hbv{T'}}{\mu T \sqcap \mu T'}
\end{mathpar}
The brand-variable rules are deliberately \emph{conservative}: a $\bv{i}$ may be
instantiated to any brand, so it costs every $\bv{j}$ and every $\bra{n}$
(unconditionally, in both directions), and two syntactically distinct $\forall$- or
$\mu$-bodies containing brand variables can become equal after instantiation
(\textsc{callbbv}/\textsc{cmubv}).  This is exactly what makes disjointness stable
under instantiation: $A * B \Rightarrow \subs{A}{\bv{d}}{\bra{n}} *
\subs{B}{\bv{d}}{\bra{n}}$, with no freshness side condition anywhere
(\textsf{disj\_bsubst}).  The price: nothing typed at a brand variable (or a
$\bv{}$-containing $\mu$/$\forall$) can be \emph{merged} with anything brand- or
$\bv{}$-typed — the same wrap-it-in-records discipline unions already impose via
\textsc{coror}.

Casting follows the union tag; at $\mu$- and $\forall$-types it is the identity on
folds of the same body resp.\ brand closures with the same annotation (the
\textsc{cwrap} pattern), and \emph{undefined} at $X_k$ and $\bv{i}$ (no cast rule —
a term whose redex casts at a brand variable is stuck until instantiated).  A fixpoint
closure carries two codomains: casts rewrite only the external $B_x$; the internal $B$
is pinned because beta reinstalls the self-reference at it.

\begin{definition}[\textsf{viewL}, \textsf{viewR}, \textsf{WrapRel}, \textsf{RawRel}, \textsf{LRg}, \textsf{LR}]
$o \in \{\mathsf{none}\} \cup \mathbb{N}\times\mathsf{Typ}\times\mathsf{Typ}$; \quad
$\eta : \mathbb{N} \to \mathsf{Exp} \to \mathsf{Exp} \to \mathsf{Prop}$; \quad stores $\Delta_1, \Delta_2$ ambient.
\[
\begin{array}{l@{\qquad}l}
\mathsf{viewL}\,\mathsf{none}\,T = T & \mathsf{viewL}\,(n,R_1,R_2)\,T = \subs{T}{n}{R_1} \\
\mathsf{viewR}\,\mathsf{none}\,T = T & \mathsf{viewR}\,(n,R_1,R_2)\,T = \subs{T}{n}{R_2}
\end{array}
\]
\[
\begin{array}{r@{\;}c@{\;}l}
\mathsf{WrapRel}\ m\ (v_1,v_2) &=& \hast{\Delta_1}{\tp}{v_1}{\bra{m}} \wedge \hast{\Delta_2}{\tp}{v_2}{\bra{m}} \wedge
  \exists w_1\,w_2.\ v_1 = \wrp{m}{w_1} \wedge v_2 = \wrp{m}{w_2} \wedge \eta\,m\,w_1\,w_2 \\
\mathsf{RawRel}\ n\ R_1\ R_2\ (v_1,v_2) &=& v_1\ \Val \wedge v_2\ \Val \wedge
  \hast{\Delta_1}{\tp}{v_1}{R_1} \wedge \hast{\Delta_2}{\tp}{v_2}{R_2} \wedge \eta\,n\,v_1\,v_2
\end{array}
\]
\[
\begin{array}{r@{\;}c@{\;}l}
\LRg{o}{\Int}(v_1,v_2) &=& \exists i.\ v_1 = i \wedge v_2 = i \\
\LRg{o}{\tp}(v_1,v_2) &=& v_1 = \tp \wedge v_2 = \tp \\
\LRg{o}{\bra{m}}(v_1,v_2) &=& \begin{cases}
    \mathsf{RawRel}\ n\ R_1\ R_2\ (v_1,v_2) & o = (n,R_1,R_2),\ m = n \\
    \mathsf{WrapRel}\ m\ (v_1,v_2) & \text{otherwise}
  \end{cases} \\
\LRg{o}{A \& B}(v_1,v_2) &=& \hast{\Delta_1}{\tp}{v_1}{\mathsf{viewL}\,o\,(A\&B)} \wedge \hast{\Delta_2}{\tp}{v_2}{\mathsf{viewR}\,o\,(A\&B)} \;\wedge \\
  && \exists a_1\,b_1\,a_2\,b_2.\ v_1 = a_1 \mrg b_1 \wedge v_2 = a_2 \mrg b_2 \wedge \LRg{o}{A}(a_1,a_2) \wedge \LRg{o}{B}(b_1,b_2) \\
\LRg{o}{\{\ell:A\}}(v_1,v_2) &=& \exists w_1\,w_2.\ v_1 = \{\ell = w_1\} \wedge v_2 = \{\ell = w_2\} \wedge \LRg{o}{A}(w_1,w_2) \\
\LRg{o}{A \to B}(v_1,v_2) &=& v_1\ \Val \wedge v_2\ \Val \wedge
  \hast{\Delta_1}{\tp}{v_1}{\mathsf{viewL}\,o\,(A{\to}B)} \wedge \hast{\Delta_2}{\tp}{v_2}{\mathsf{viewR}\,o\,(A{\to}B)} \;\wedge \\
  && \forall u_1\,u_2.\ \LRg{o}{A}(u_1,u_2) \Rightarrow \forall \rho_1\,\rho_2.\ \rho_1\ \Val \Rightarrow \rho_2\ \Val \Rightarrow \\
  && \quad \exists w_1\,w_2.\ \msteps{\rho_1}{v_1\,u_1}{w_1} \wedge \msteps{\rho_2}{v_2\,u_2}{w_2} \wedge \LRg{o}{B}(w_1,w_2)
\end{array}
\]
\[
\begin{array}{r@{\;}c@{\;}l}
\LRg{o}{A \vee B}(v_1,v_2) &=& \bigl(\exists w_1\,w_2.\ v_1 = \mathsf{inl}^{\mathsf{viewL}\,o\,B} w_1 \wedge v_2 = \mathsf{inl}^{\mathsf{viewR}\,o\,B} w_2 \wedge \LRg{o}{A}(w_1,w_2)\bigr) \;\vee \\
  && \bigl(\exists w_1\,w_2.\ v_1 = \mathsf{inr}^{\mathsf{viewL}\,o\,A} w_1 \wedge v_2 = \mathsf{inr}^{\mathsf{viewR}\,o\,A} w_2 \wedge \LRg{o}{B}(w_1,w_2)\bigr) \\
\LRg{o}{X_k}(v_1,v_2) &=& \bot \qquad\qquad \LRg{o}{\bv{i}}(v_1,v_2) \;=\; \bot \\
\LRg{o}{\mu T}(v_1,v_2) &=& v_1\ \Val \wedge v_2\ \Val \wedge
  \hast{\Delta_1}{\tp}{v_1}{\mathsf{viewL}\,o\,(\mu T)} \wedge \hast{\Delta_2}{\tp}{v_2}{\mathsf{viewR}\,o\,(\mu T)} \wedge v_1 = v_2 \\
\LRg{o}{\allb{A}}(v_1,v_2) &=& v_1\ \Val \wedge v_2\ \Val \wedge
  \hast{\Delta_1}{\tp}{v_1}{\mathsf{viewL}\,o\,(\allb{A})} \wedge \hast{\Delta_2}{\tp}{v_2}{\mathsf{viewR}\,o\,(\allb{A})} \;\wedge \\
  && \forall m.\ \forall \rho_1\,\rho_2.\ \rho_1\ \Val \Rightarrow \rho_2\ \Val \Rightarrow \\
  && \quad \exists w_1\,w_2.\ \msteps{\rho_1}{\bapp{v_1}{m}}{w_1} \wedge \msteps{\rho_2}{\bapp{v_2}{m}}{w_2} \wedge \LRg{o}{\subs{A}{\bv{0}}{\bra{m}}}(w_1,w_2)
\end{array}
\]
\[
\LRc{T} \;:=\; \LRg{\mathsf{none}}{T}
\]
The $\forall$-clause is \emph{instantiation closure}: related generic values yield, at
every concrete brand $m$, related runs of the instantiation at the substituted body
type.  $\eta$ is never extended and no freshness is demanded — at brand-$m$ positions
the relation is whatever $\eta$ already says about $m$, which is exactly what
representation independence needs (the client instantiates at the provider's own
allocated brand, where $\eta$ relates the two representations).  Typings stay at the
\emph{syntactic} type, so degenerate $\bv{}$-annotated values relate vacuously through
the $\bot$ clause at $\bv{i}$.  The recursion descends on $\mathsf{tsize}$
($\subs{A}{\bv{0}}{\bra{m}}$ is not a subterm but is size-preserving), so the
definition is by well-founded recursion.  $\mu$ stays opaque (a finer clause needs step
indexing); free theorems over \emph{arbitrary} relations at $\beta$ (relational
parametricity proper) are future work.
\end{definition}

\begin{definition}[\textsf{Finitary}]
$\mathsf{Finitary}\ e$: the least predicate closed under every term former
\emph{except} the fixpoints $\mu\lambda(A{\to}B).e$ and their closures.  Generic terms
$\Lambda$/$\langle\cdot,\Lambda\rangle$/$\bapp{\cdot}{n}$ \emph{are} included:
$\forall$-beta rewrites only annotations and never duplicates subterms, so the fragment
stays normalizing.  The relational layer below is stated for this fragment; the safety
layer (determinism, progress, preservation) covers the full language.
\end{definition}

\hast{\emptyset}{\Gamma}{e}{A} \;\wedge\; \mathsf{Finitary}\ e \;\Longrightarrow\; \mathsf{SemTyp}\ \Delta_1\ \Delta_2\ \eta\ \Gamma\ e\ e\ A

Proved by strong induction on $\mathsf{esize}\ e$: the $\Lambda$-cases recurse at the
\emph{instantiated} body — the unsubstituted body cannot run (casts at $\bv{}$
annotations are stuck) — whose typing comes from the instantiation lemma
$\hast{\Delta}{\Gamma}{e}{A} \Rightarrow
\hast{\Delta}{\subs{\Gamma}{\bv{d}}{\bra{n}}}{\subs{e}{\bv{d}}{\bra{n}}}{\subs{A}{\bv{d}}{\bra{n}}}$
(\textsf{hastype\_bsubst}), not from a subderivation.
\end{theorem}

\begin{theorem}[\textsf{normalization}]
\[
\hast{\emptyset}{\tp}{e}{A} \;\wedge\; \mathsf{Finitary}\ e \;\Longrightarrow\; \exists v.\ v\ \Val \;\wedge\; \msteps{\tp}{e}{v}
\]
\end{theorem}

\begin{theorem}[\textsf{sealing}]
\[
\LRc{A}(p_1',p_2') \;\wedge\; \hast{\emptyset}{A}{e}{\Int} \;\wedge\; \mathsf{Finitary}\ e \;\Longrightarrow\;
\exists i.\ \msteps{p_1'}{e}{i} \;\wedge\; \msteps{p_2'}{e}{i}
\]
\end{theorem}

\begin{theorem}[\textsf{coe\_lr}]
\[
\mathsf{OpenBrand}\ \Delta_1\ \Delta_2\ \eta\ n\ R_1\ R_2 \;\wedge\;
\mathsf{WfSig}\ n\ R_1\ S \;\wedge\; \mathsf{WfSig}\ n\ R_2\ S \;\Longrightarrow
\]
\[
\begin{array}{l}
\bigl(\forall v_1\,v_2\,w_1\,w_2.\
  \LRg{(n,R_1,R_2)}{S}(v_1,v_2) \Rightarrow \sealv{n}{R_1}{S}{v_1}{w_1} \Rightarrow \sealv{n}{R_2}{S}{v_2}{w_2} \Rightarrow \LRc{S}(w_1,w_2)\bigr) \;\wedge \\
\bigl(\forall v_1\,v_2\,w_1\,w_2.\
  \LRc{S}(v_1,v_2) \Rightarrow \unsealv{n}{R_1}{S}{v_1}{w_1} \Rightarrow \unsealv{n}{R_2}{S}{v_2}{w_2} \Rightarrow \LRg{(n,R_1,R_2)}{S}(w_1,w_2)\bigr)
\end{array}
\]
\end{theorem}

\begin{theorem}[\textsf{representation\_independence}]
\[
\begin{array}{l}
\mathsf{OpenBrand}\ \Delta_1\ \Delta_2\ \eta\ n\ R_1\ R_2 \;\wedge\;
\mathsf{WfSig}\ n\ R_1\ S \;\wedge\; \mathsf{WfSig}\ n\ R_2\ S \;\wedge \\
\LRg{(n,R_1,R_2)}{S}(p_1,p_2) \;\wedge\; \hast{\emptyset}{S}{e}{\Int} \;\wedge\; \mathsf{Finitary}\ e \\
\Longrightarrow\;
\exists i.\ \msteps{\tp}{(\sealE{n}{R_1}{S}{p_1}) \bx e}{i} \;\wedge\; \msteps{\tp}{(\sealE{n}{R_2}{S}{p_2}) \bx e}{i}
\end{array}
\]
\end{theorem}

\begin{theorem}[\textsf{generic\_client\_representation\_independence}]
One \emph{generic} client, typed once at $\allb{(S_\beta \to \Int)}$ with the signature
abstract in the brand it will be instantiated at, observes the same integer against two
sealed providers of that signature over different representations.  Writing
$S^{*} := \subs{S_\beta}{\bv{0}}{\bra{n}}$:
\[
\begin{array}{l}
\mathsf{OpenBrand}\ \Delta_1\ \Delta_2\ \eta\ n\ R_1\ R_2 \;\wedge\;
\mathsf{WfSig}\ n\ R_1\ S^{*} \;\wedge\; \mathsf{WfSig}\ n\ R_2\ S^{*} \;\wedge \\
\LRg{(n,R_1,R_2)}{S^{*}}(p_1,p_2) \;\wedge\;
\hast{\emptyset}{\tp}{c_t}{\allb{(S_\beta \to \Int)}} \;\wedge\; \mathsf{Finitary}\ c_t \\
\Longrightarrow\;
\exists i.\ \msteps{\tp}{(\bapp{c_t}{n})\,(\sealE{n}{R_1}{S^{*}}{p_1})}{i} \;\wedge\;
\msteps{\tp}{(\bapp{c_t}{n})\,(\sealE{n}{R_2}{S^{*}}{p_2})}{i}
\end{array}
\]
The client instantiates at the provider's \emph{own} allocated brand $n$: the
$\forall$-clause needs no freshness, and $\eta$ at $n$ is exactly the relation between
the two representations, so $\bapp{\cdot}{n}$ lands the client in the
concrete-signature world where \textsf{representation\_independence} already lives.
\end{theorem}

\begin{theorem}[\textsf{seal\_type\_preservation}]
\[
\elab{\Delta}{\Gamma}{e}{A}{\overline{e}} \;\Longrightarrow\; \hast{\sT{\Delta}}{\sT{\Gamma}}{\overline{e}}{\sT{A}}
\]
\end{theorem}

\begin{theorem}[\textsf{elabSeal\_weaken\_store}]
\[
\Delta \storele \Delta' \;\wedge\; \elab{\Delta}{\Gamma}{e}{A}{\overline{e}} \;\Longrightarrow\; \elab{\Delta'}{\Gamma}{e}{A}{\overline{e}}
\]
\end{theorem}

\begin{theorem}[\textsf{elabSeal\_uniqueness}]
Elaboration is a function of the term and (for non-values) its context:
\[
\elab{\Delta}{\Gamma_1}{e}{T_1}{\overline{e}_1} \;\wedge\; \elab{\Delta}{\Gamma_2}{e}{T_2}{\overline{e}_2}
\;\wedge\; (\Gamma_1 = \Gamma_2 \vee e\ \Val) \;\Longrightarrow\; T_1 = T_2 \wedge \overline{e}_1 = \overline{e}_2
\]
\end{theorem}

\begin{theorem}[\textsf{source\_sprogress}, \textsf{source\_spreservation}, \textsf{source\_bigstep\_deterministic}]
Pure source metatheory, with elaborability playing the role of typing (SStep is the
source small-step, BStep the big-step):
\[
\begin{array}{l}
(\exists \overline{e}.\ \elab{\Delta}{\tp}{e}{A}{\overline{e}}) \;\Longrightarrow\; e\ \Val \;\vee\; \exists e'.\ \mathsf{SStep}\ \tp\ e\ e' \\[2pt]
(\exists \overline{e}.\ \elab{\Delta}{\tp}{e}{A}{\overline{e}}) \;\wedge\; \mathsf{SStep}\ \tp\ e\ e' \;\Longrightarrow\; \exists \overline{e}'.\ \elab{\Delta}{\tp}{e'}{A}{\overline{e}'} \\[2pt]
(\exists \overline{e}.\ \elab{\Delta}{\tp}{e}{A}{\overline{e}}) \;\wedge\; \bstep{\tp}{e}{v_1} \;\wedge\; \bstep{\tp}{e}{v_2} \;\Longrightarrow\; v_1 = v_2
\end{array}
\]
(Each also holds generalized over a well-typed environment; sealed-unit runs are
covered — the sealing cases of preservation and progress go through the source
coercions.)
\end{theorem}

\begin{theorem}[\textsf{seal\_separate\_compilation}]
